# Auth & Data Rebuild Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace Supabase Auth with Better Auth + Cloudflare Access, migrate GitHub-JSON data to Postgres via Drizzle, flatten repo to root-level Next.js, adopt Bun, deploy on Dokploy with self-hosted Supabase.

**Architecture:** Single Next.js app on Bun. CF Access at the edge injects a verified JWT; `middleware.ts` validates it with `jose` and loads a Better Auth session from Postgres. All financial data lives in Postgres accessed through Drizzle. Cron jobs are API routes triggered by Dokploy scheduler over localhost.

**Tech Stack:** Bun, Next.js 16, TypeScript, Better Auth (admin plugin + access control), Drizzle ORM + drizzle-kit, postgres-js, jose (JWT verification), Zod, Supabase (self-hosted Postgres + Studio on Dokploy), Supabase CLI (local dev), Dokploy (deployment + cron).

**Spec:** [`docs/superpowers/specs/2026-04-20-auth-and-data-rebuild-design.md`](../specs/2026-04-20-auth-and-data-rebuild-design.md)

```ts
    walletAddress: text('wallet_address').notNull().references(() => wallet.address),

    ixWalletTime: index('tx_wallet_time_idx').on(t.walletAddress, t.blockTime),
```

```ts
import { jwtVerify, createLocalJWKSet, type JWTPayload } from 'jose'

export type CfClaims = JWTPayload & {
  email: string
  identity_nonce?: string
  sub?: string
}

type VerifyOptions = {
  aud: string
  issuer: string
  jwks: () => Promise<{ keys: any[] }>
}

export async function verifyCfJwt(token: string, opts: VerifyOptions): Promise<CfClaims> {
  const jwks = await opts.jwks()
  const keySet = createLocalJWKSet(jwks as any)

  const { payload } = await jwtVerify(token, keySet, {
    audience: opts.aud,
    issuer: opts.issuer,
  })

  if (typeof payload.email !== 'string' || payload.email.length === 0) {
    throw new Error('CF JWT missing email claim')
  }

  return payload as CfClaims
}

// Production JWKS fetcher with in-memory cache (5 min TTL)
let cachedJwks: { fetchedAt: number; data: { keys: any[] } } | null = null

export async function fetchCfJwks(teamDomain: string): Promise<{ keys: any[] }> {
  const now = Date.now()
  if (cachedJwks && now - cachedJwks.fetchedAt < 5 * 60 * 1000) {
    return cachedJwks.data
  }
  const url = `https://${teamDomain}/cdn-cgi/access/certs`
  const res = await fetch(url)
  if (!res.ok) throw new Error(`CF JWKS fetch failed: ${res.status}`)
  const data = await res.json() as { keys: any[] }
  cachedJwks = { fetchedAt: now, data }
  return data
}
```

```ts
  if (!perms) return false
```

```ts
export type SkipAuthMode = 'none' | 'admin' | 'viewer'

export function assertSafeStartup(env: { NODE_ENV?: string; SKIP_AUTH?: string }): void {
  const skip = env.SKIP_AUTH ?? 'none'
  if (!['none', 'admin', 'viewer'].includes(skip)) {
    throw new Error(`invalid SKIP_AUTH value: ${skip}. Expected 'none' | 'admin' | 'viewer'.`)
  }
  if (skip !== 'none' && env.NODE_ENV !== 'development') {
    throw new Error(
      `SKIP_AUTH=${skip} is only allowed when NODE_ENV=development. Refusing to boot.`
    )
  }
}

export function readSkipAuthMode(): SkipAuthMode {
  assertSafeStartup(process.env)
  return (process.env.SKIP_AUTH ?? 'none') as SkipAuthMode
}
```

```ts
  email: string
  role: Role | null
  departments: string[]

export async function loadUserByEmail(email: string): Promise<RequestUser | null> {
  const lower = email.toLowerCase()
  // Look up account by its email (multi-email via account linking)
  const [acc] = await db.select({ userId: account.userId })
    .from(account)
    .where(eq(account.email, lower))
    .limit(1)

  let userId: string | null = acc?.userId ?? null

  // Fallback: direct user.email match (for users not yet linked)
  if (!userId) {
    const [u] = await db.select({ id: user.id })
      .from(user)
      .where(eq(user.email, lower))
      .limit(1)
    userId = u?.id ?? null
  }

  if (!userId) return null

  const [row] = await db.select().from(user).where(eq(user.id, userId)).limit(1)
  if (!row) return null

  const depts = await db.select({ department: userDepartment.department })
    .from(userDepartment)
    .where(eq(userDepartment.userId, userId))

  return {
    id: row.id,
    email: row.email,
    role: (row.role as Role | null) ?? null,
    departments: depts.map(d => d.department),
  }
}
```

```ts
export async function middleware(request: NextRequest) {
  const { pathname } = request.nextUrl

  // Cron routes: authorized via CRON_SECRET in route handler, skip auth here
  if (pathname.startsWith('/api/cron/')) {

  if (PUBLIC_PATHS.has(pathname)) {
    return NextResponse.next()

    }
  } else {
    // Production: verify CF JWT
    const token = request.headers.get(CF_JWT_HEADER)
    if (!token) {
      return denyResponse(request, 401, 'auth.no_cf_jwt', 'Missing Cloudflare Access JWT.')
    }
    let claims
    try {
      claims = await verifyCfJwt(token, {
        aud: process.env.CF_ACCESS_AUD!,
        issuer: `https://${process.env.CF_ACCESS_TEAM_DOMAIN!}`,
        jwks: () => fetchCfJwks(process.env.CF_ACCESS_TEAM_DOMAIN!),
      })
    } catch (err) {
      await logAudit({
        action: 'access.denied.invalid_jwt',
        metadata: { reason: (err as Error).message, path: pathname },
      })
      return denyResponse(request, 401, 'auth.invalid_jwt', 'Invalid Cloudflare Access JWT.')
    }

    const email = (claims.email ?? '').toLowerCase()
    user = await loadUserByEmail(email)
    if (!user) {
      await logAudit({
        action: 'access.denied.unknown_email',
        metadata: { email, path: pathname },
      })
      return pathname.startsWith('/api/')
        ? NextResponse.json({ error: 'Unknown email', code: 'auth.unknown_email' }, { status: 403 })
        : NextResponse.redirect(new URL('/no-access', request.url))
    }
  }

  if (!user) {
    return denyResponse(request, 401, 'auth.no_user', 'No authenticated user.')
  }

  if (!user.role) {
    await logAudit({
      userId: user.id === 'dev-user' ? null : user.id,
      action: 'access.denied.no_role',
      metadata: { email: user.email, path: pathname },
    })

  const response = NextResponse.next()
```

```ts
  const h = await headers()
  const id = h.get('x-user-id')

  return user
}
```

```ts
import { NextResponse } from 'next/server'
import { z } from 'zod'
import { and, eq, inArray } from 'drizzle-orm'
import { db } from '@/lib/db'
import { user, account } from '@/lib/db/schema/auth'
import { userDepartment } from '@/lib/db/schema/authz'
import { requirePermission } from '@/lib/auth/require-permission'
import { logAudit } from '@/lib/audit'

const UpdateUserSchema = z.object({
  role: z.enum(['admin', 'budget_viewer']).nullable().optional(),
  departments: z.array(z.string()).optional(),
  banned: z.boolean().optional(),
  linkEmail: z.string().email().optional(),
  unlinkEmail: z.string().email().optional(),
})

export async function PATCH(req: Request, { params }: { params: Promise<{ id: string }> }) {
  const { user: me, deny } = await requirePermission('admin:manage_users')
  if (deny) return deny

  const { id } = await params
  const body = UpdateUserSchema.parse(await req.json())

  const [existing] = await db.select().from(user).where(eq(user.id, id))
  if (!existing) {
    return NextResponse.json({ error: 'User not found', code: 'admin.user_not_found' }, { status: 404 })
  }

  const changes: Record<string, unknown> = {}

  if ('role' in body) {
    await db.update(user).set({ role: body.role ?? null, updatedAt: new Date() }).where(eq(user.id, id))
    await logAudit({
      userId: me.id,
      action: 'role.assigned',
      resourceType: 'user',
      resourceId: id,
      metadata: { previous: existing.role, new: body.role ?? null },
    })
    changes.role = body.role ?? null
  }

  if (body.departments) {
    await db.delete(userDepartment).where(eq(userDepartment.userId, id))
    if (body.departments.length > 0) {
      await db.insert(userDepartment).values(body.departments.map(d => ({ userId: id, department: d })))
    }
    await logAudit({
      userId: me.id,
      action: 'departments.assigned',
      resourceType: 'user',
      resourceId: id,
      metadata: { departments: body.departments },
    })
    changes.departments = body.departments
  }

  if ('banned' in body) {
    await db.update(user).set({ banned: body.banned!, updatedAt: new Date() }).where(eq(user.id, id))
    await logAudit({
      userId: me.id,
      action: body.banned ? 'user.banned' : 'user.unbanned',
      resourceType: 'user',
      resourceId: id,
    })
    changes.banned = body.banned
  }

  if (body.linkEmail) {
    // Create an 'email-alias' provider account row
    const email = body.linkEmail.toLowerCase()
    await db.insert(account).values({
      id: crypto.randomUUID(),
      userId: id,
      accountId: email,
      providerId: 'email-alias',
      email,
    })
    await logAudit({
      userId: me.id,
      action: 'account.linked',
      resourceType: 'user',
      resourceId: id,
      metadata: { email },
    })
    changes.linkedEmail = email
  }

  if (body.unlinkEmail) {
    const email = body.unlinkEmail.toLowerCase()
    await db.delete(account).where(and(eq(account.userId, id), eq(account.email, email), eq(account.providerId, 'email-alias')))
    await logAudit({
      userId: me.id,
      action: 'account.unlinked',
      resourceType: 'user',
      resourceId: id,
      metadata: { email },
    })
    changes.unlinkedEmail = email
  }

  return NextResponse.json({ ok: true, changes })
}
```

```ts
  if (a.length !== b.length) return false
  let result = 0
  for (let i = 0; i < a.length; i++) result |= a.charCodeAt(i) ^ b.charCodeAt(i)
  return result === 0
}

export async function runCron(
  name: string,
  request: Request,
  job: () => Promise<JobResult>,
): Promise<Response> {
  const expected = process.env.CRON_SECRET
  if (!expected) {
    return NextResponse.json({ error: 'CRON_SECRET not configured', code: 'cron.misconfigured' }, { status: 500 })
  }

  const auth = request.headers.get('authorization') ?? ''
  const provided = auth.startsWith('Bearer ') ? auth.slice(7) : ''
  if (!provided || !constantTimeEquals(provided, expected)) {
    return NextResponse.json({ error: 'Unauthorized', code: 'cron.unauthorized' }, { status: 401 })
  }

  const startedAt = Date.now()
  try {
    const result = await job()
    await logAudit({
      action: `cron.${name}.succeeded`,
      metadata: { durationMs: Date.now() - startedAt, ...result },
    })
    return NextResponse.json(result)
  } catch (err) {
    const msg = (err as Error).message
    await logAudit({
      action: `cron.${name}.failed`,
      metadata: { durationMs: Date.now() - startedAt, error: msg },
    })
    return NextResponse.json(
      { ok: false, error: msg, code: 'cron.failed' },
      { status: 500 },
    )
  }
}
```

```ts
  const entities = await db.select().from(entity).where(eq(entity.isActive, true))
  const now = new Date()

    const month = now.getUTCMonth()

      { reportType: 'balance_sheet', periodStart: today, periodEnd: today, filterParams: {} },

        periodStart: monthStart,

        eq(qbReportCache.reportType, r.reportType),
        eq(qbReportCache.periodStart, r.periodStart),
        eq(qbReportCache.periodEnd, r.periodEnd),
      )).limit(1)

      if (existing[0] && now.getTime() - existing[0].fetchedAt.getTime() < CACHE_STALE_MS) {
        continue

        start_date: r.periodStart,
        end_date: r.periodEnd,
        accounting_method: 'Accrual',
        minorversion: '65',

      await db.insert(qbReportCache).values({
```

```ts
  entity: z.string(),
  periodStart: z.string(),
  periodEnd: z.string(),
  account: z.string().optional(),
})

export async function GET(req: Request, { params }: { params: Promise<{ type: string }> }) {
  const { user, deny } = await requirePermission('qb:read')
  if (deny) return deny

  const { type } = await params
  if (!['balance_sheet', 'profit_and_loss', 'general_ledger'].includes(type)) {
    return NextResponse.json({ error: 'Invalid report type', code: 'qb.invalid_type' }, { status: 400 })
  }

  const url = new URL(req.url)
  const q = QuerySchema.parse(Object.fromEntries(url.searchParams))
  const filterParams = q.account ? { account: q.account } : {}

  const [row] = await db.select().from(qbReportCache).where(and(
    eq(qbReportCache.entityId, q.entity),
    eq(qbReportCache.reportType, type as any),
    eq(qbReportCache.periodStart, q.periodStart),
    eq(qbReportCache.periodEnd, q.periodEnd),
  )).limit(1)

  if (!row) {
    return NextResponse.json({ error: 'Report not cached yet. Cron runs hourly.', code: 'qb.not_cached' }, { status: 404 })
  }

  return NextResponse.json({ data: row.data, fetchedAt: row.fetchedAt, stale: false })
}
```

# Implementation notes for executing agents

- **Each task is independently reviewable.** Between tasks, the tree should typecheck and tests should pass. If a task leaves the tree broken, split it further.
- **Never skip the commit step.** Frequent commits protect against rebase hell.
- **If a test fails unexpectedly**, read the error carefully before modifying test or code. Drizzle errors can be cryptic — usually a missing FK row or wrong column name.
- **SQL only appears in `scripts/apply-triggers.ts`** — nowhere else. If you find yourself writing `CREATE TABLE` or `INSERT INTO` in TypeScript, stop and use Drizzle's query builder instead.
- **CF JWT claims vary by IdP**; the `email` claim is reliable from Google Workspace. If another IdP is used later, the claim path may need adjustment in `verifyCfJwt`.
- **Better Auth is pinned**; do not upgrade it during this rebuild. Pin the exact version in `package.json`.
